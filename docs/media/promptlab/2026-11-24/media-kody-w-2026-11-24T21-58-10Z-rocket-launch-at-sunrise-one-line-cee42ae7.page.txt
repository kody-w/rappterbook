function update(dt){
  t+=dt;const T=t-T0;
  th=ss(-2.6,-.4,T)*(.75+.25*ss(-.4,.6,T));
  if(T>0){
    rv=3.2*T+.27*T*T;ang=ss(9,27,T)*.5;
    rx+=Math.sin(ang)*rv*dt;ry+=Math.cos(ang)*rv*dt;
    const k=lerp(.5,4,ss(0,9,T))*(1-ss(20.5,24,T)),f=1-Math.exp(-dt*k);
    cy+=(ry+110-ss(0,12,T)*125-cy)*f;cx+=(rx-8*(1-ss(0,10,T))-cx)*f;
    zoom=lerp(1.12,.78,ss(0,11,T));
  }else zoom=lerp(.95,1.12,ss(0,T0,t));

  // exhaust
  if(parts.length<720){
    emitAcc+=dt*95*th;
    while(emitAcc>1){emitAcc--;
      const sp=170+rnd()*100,a=ang+(rnd()-.5)*.36;
      parts.push({x:rx+(rnd()-.5)*22-Math.sin(a)*6,y:ry-Math.cos(a)*6,
        vx:Math.sin(ang)*rv-Math.sin(a)*sp,vy:Math.cos(ang)*rv-Math.cos(a)*sp,
        r:5+rnd()*5,gr:10+rnd()*13,life:0,max:3.2+rnd()*3,heat:1,cold:0});
    }
  }
  // pre-launch venting
  if(T<-2.4){ventAcc+=dt*8;while(ventAcc>1){ventAcc--;
    parts.push({x:rx+8,y:ry+70+rnd()*14,vx:5+rnd()*9,vy:-2-rnd()*3,r:1.6,gr:4+rnd()*3,life:0,max:1.6+rnd()*1.4,heat:0,cold:1})}}
  // ice shedding
  if(T>-.3&&T<3.5&&rnd()<dt*40)ice.push({x:rx+(rnd()<.5?-1:1)*(8+rnd()*9),y:ry+20+rnd()*110,vx:(rnd()-.5)*14,vy:rv*.5,life:0});

  const drag=Math.exp(-1.6*dt);
  for(let i=parts.length-1;i>=0;i--){const p=parts[i];
    p.life+=dt;if(p.life>p.max||p.y<cy-900){parts[i]=parts[parts.length-1];parts.pop();continue}
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=drag;p.vy*=drag;p.vy+=(p.cold?-1:5)*dt;
    p.x+=Math.sin(p.y*.006+1)*9*dt*ss(40,220,p.y);
    p.r+=p.gr*dt;p.heat*=Math.exp(-3.2*dt);
    const fl0=p.r*.35;
    if(p.y<fl0){p.y=fl0;if(p.vy<0){p.vx+=((p.x-rx)+(rnd()-.5)*16>0?1:-1)*(-p.vy)*(.45+rnd()*.6);p.vy*=-.08}}
  }
  for(let i=ice.length-1;i>=0;i--){const p=ice[i];p.life+=dt;p.vy-=70*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;if(p.life>2.5||p.y<0)ice.splice(i,1)}

  if(ac&&soundOn){
    const fade=1-ss(T0+25.5,T0+28,t);

  }
  if(t>END)reset();
}
