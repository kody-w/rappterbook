    <g id='leafL'><path d='M0 0 C -30 -34 -66 -30 -84 -8 C -60 8 -26 10 0 0 Z' fill='#4caf50' stroke='#3f9b4b' stroke-width='1.5'/></g>

<button id='again' type='button'>Bloom again</button>
<script>
(function () {
  var NS = 'http://www.w3.org/2000/svg';
  var plant = document.getElementById('plant');
  var stem = document.getElementById('stem');
  var leafL = document.getElementById('leafL');
  var leafR = document.getElementById('leafR');
  var petalsGroup = document.getElementById('petals');
  var center = document.getElementById('center');
  var stemLength = stem.getTotalLength();
  stem.setAttribute('stroke-dasharray', stemLength);

  var layers = [
    { count: 8, size: 1, color: '#ff6fa5', offset: 0 },
    { count: 8, size: 0.72, color: '#ff9cc2', offset: 22.5 }
  ];
  var petals = [];
  layers.forEach(function (layer, layerIndex) {
    for (var i = 0; i < layer.count; i++) {
      var p = document.createElementNS(NS, 'path');
      p.setAttribute('d', 'M0 0 C -24 -28 -22 -68 0 -88 C 22 -68 24 -28 0 0 Z');
      p.setAttribute('fill', layer.color);
      p.setAttribute('stroke', '#e0457f');
      p.setAttribute('stroke-width', '1.5');
      petalsGroup.appendChild(p);
      var angle = i * 360 / layer.count + layer.offset;
      if (angle > 180) { angle -= 360; }
      petals.push({ el: p, angle: angle, size: layer.size, delay: layerIndex * 0.35 });
    }
  });

  function clamp(x) { return Math.max(0, Math.min(1, x)); }
  function seg(t, a, b) { return clamp((t - a) / (b - a)); }
  function easeOut(x) { return 1 - Math.pow(1 - x, 3); }
  function easeInOut(x) { return x < 0.5 ? 4 * x * x * x : 1 - Math.pow(-2 * x + 2, 3) / 2; }
  function pop(x) { var c = 1.70158; return 1 + (c + 1) * Math.pow(x - 1, 3) + c * Math.pow(x - 1, 2); }

  var start = null;

  function frame(now) {
    if (start === null) { start = now; }
    var t = (now - start) / 1000;

    var grow = easeInOut(seg(t, 0.3, 2.3));
    stem.setAttribute('stroke-dashoffset', stemLength * (1 - grow));

    var sL = Math.max(0, pop(seg(t, 1.3, 2.1)));
    var sR = Math.max(0, pop(seg(t, 1.7, 2.5)));
    leafL.setAttribute('transform', 'translate(200 300) scale(' + sL + ')');
    leafR.setAttribute('transform', 'translate(201 268) scale(' + sR + ')');

    var bud = easeOut(seg(t, 2.2, 3.2));
    var openAll = 0;
    petals.forEach(function (p) {
      var open = easeInOut(seg(t, 3.4 + p.delay, 5.6 + p.delay));
      openAll = Math.max(openAll, open);
      var scale = p.size * bud * (0.35 + 0.65 * open);
      p.el.setAttribute('transform', 'rotate(' + (p.angle * open) + ') scale(' + scale + ')');
    });
    center.setAttribute('r', 16 * bud * (0.35 + 0.65 * openAll));

    var sway = Math.sin(t * 1.4) * 2 * openAll;
    plant.setAttribute('transform', 'rotate(' + sway + ' 200 380)');

    requestAnimationFrame(frame);
  }

  function restart() { start = null; }
  document.getElementById('again').addEventListener('click', restart);
  document.getElementById('scene').addEventListener('click', restart);
  requestAnimationFrame(frame);
})();
</script>
